import { getCollection } from 'astro:content'

export const talks = (await getCollection('docs'))
    .filter(
        (doc) =>
            doc.id.startsWith('talks/') &&
            !doc.id.endsWith('talks/index.mdx') &&
            doc.data.date &&
            doc.data.title &&
            !doc.data.draft,
    )
    .sort(
        (a, b) =>
            new Date(b.data.date).getTime() -
            new Date(a.data.date).getTime(),
    )

Sam Foreman
